# --- Core API Framework ---
# fastapi bumped 0.115.0 -> 0.141.1: no era compatible con starlette
# parcheado (pinback a starlette<0.39.0, y las CVEs de starlette — DoS en
# multipart, Host header sin validar en request.url — solo se arreglan del
# todo en starlette>=1.0.1/1.1.0/1.3.0/1.3.1). 0.141.1 exige
# starlette>=0.46.0 sin techo, así que pip resuelve el 1.3.1 más reciente
# automáticamente sin pin explícito aquí.
fastapi==0.141.1
# Pin explícito: fastapi solo exige starlette>=0.46.0 (sin techo), y pip no
# sube más allá del mínimo que satisface esa cota si no se lo pedimos —
# sin este pin, un install limpio se habría quedado en 0.50.0 (todavía
# vulnerable) en vez de la 1.3.1 que arregla el resto de CVEs de starlette.
starlette==1.3.1
uvicorn[standard]==0.32.0
python-dotenv==1.2.2           # CVE fix, sin cambios de API relevantes para nosotros

# --- LangChain Core & RAG ---
# Bumps dentro de la misma rama 0.3.x (no 1.x): 1.0 reescribió la API de
# chains/agents y no lo necesitamos — estos son solo backports de seguridad.
#
# Quedan 4 CVEs sin arreglar en esta rama (solo tienen fix en la 1.x, ver
# `pip-audit -r requirements.txt`) — ninguno aplica a este código, no usamos
# esas APIs (confirmado con grep, cero resultados):
#   - langchain: path traversal en el middleware de file-search de agentes
#     (no usamos langchain.agents)
#   - langchain-core: path traversal en load_prompt()/load_prompt_from_config()
#     (no usamos load_prompt en ningún sitio, los prompts se construyen como
#     strings directamente en ollama_adapter.py)
#   - langchain-core: SSRF en ChatOpenAI.get_num_tokens_from_messages()
#     (no usamos ChatOpenAI — Ollama vía langchain_ollama, Groq vía el
#     cliente oficial `groq`, ninguno pasa por langchain_openai)
#   - langchain-text-splitters: SSRF en HTMLHeaderTextSplitter.split_text_from_url()
#     (usamos RecursiveCharacterTextSplitter, no HTMLHeaderTextSplitter)
langchain==0.3.30
langchain-community==0.3.31
langchain-core==0.3.85
langchain-text-splitters==0.3.11

# --- Vector Database ---
# 0.5.20 -> 1.5.9: Chroma Cloud corre una versión de servidor más nueva que
# ya no es compatible con el formato de respuesta que este cliente antiguo
# esperaba al listar colecciones (KeyError: '_type' en list_collections(),
# reproducido y confirmado contra la instancia real). La imagen Docker de
# ChromaDB local (docker-compose.yml) se subió a la misma versión por lo
# mismo — y su endpoint de healthcheck cambió de /api/v1/heartbeat (410
# Gone en 1.5.9) a /api/v2/heartbeat, actualizado en docker-compose.yml y
# scripts/verify_setup.py.
chromadb==1.5.9              # ChromaDB client (local Docker + Chroma Cloud, mismo cliente)
# PYSEC-2026-311: RCE pre-auth en el SERVIDOR de ChromaDB (no en este
# cliente pip) vía trust_remote_code=true al crear una colección con una
# función de embedding basada en modelo remoto — sin versión parcheada
# todavía. Nuestro código nunca pasa trust_remote_code (ver
# chromadb_adapter.py), pero la mitigación real es de red, no de código:
# el servidor de ChromaDB local no tiene auth propia, así que no debe
# quedar expuesto — ver el bind a 127.0.0.1 en docker-compose.yml. En
# prod usamos Chroma Cloud (gestionado por Chroma, fuera de nuestro
# control directo), no esta imagen.
