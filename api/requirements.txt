# --- Core API Framework ---
# fastapi bumped 0.115.0 -> 0.141.1: it wasn't compatible with a patched
# starlette (was pinned back to starlette<0.39.0, and starlette's CVEs —
# multipart DoS, unvalidated Host header in request.url — are only fully
# fixed in starlette>=1.0.1/1.1.0/1.3.0/1.3.1). 0.141.1 requires
# starlette>=0.46.0 with no ceiling, so pip resolves the latest 1.3.1
# automatically with no explicit pin needed here.
fastapi==0.141.1
# Explicit pin: fastapi only requires starlette>=0.46.0 (no ceiling), and
# pip won't go past the minimum that satisfies that bound unless we ask it
# to — without this pin, a clean install would have landed on 0.50.0 (still
# vulnerable) instead of the 1.3.1 that fixes the rest of starlette's CVEs.
starlette==1.3.1
uvicorn[standard]==0.32.0

# --- LangChain Core & RAG ---
# Bumps within the same 0.3.x branch (not 1.x): 1.0 rewrote the
# chains/agents API and we don't need that — these are just security
# backports.
#
# 4 CVEs remain unfixed on this branch (only fixed on 1.x, see
# `pip-audit -r requirements.txt`) — none of them apply to this code, we
# don't use those APIs (confirmed with grep, zero hits):
#   - langchain: path traversal in the agents' file-search middleware
#     (we don't use langchain.agents)
#   - langchain-core: path traversal in load_prompt()/load_prompt_from_config()
#     (we never use load_prompt anywhere, prompts are built as plain
#     strings directly in ollama_adapter.py)
#   - langchain-core: SSRF in ChatOpenAI.get_num_tokens_from_messages()
#     (we don't use ChatOpenAI — Ollama via langchain_ollama, Groq via the
#     official `groq` client, neither goes through langchain_openai)
#   - langchain-text-splitters: SSRF in HTMLHeaderTextSplitter.split_text_from_url()
#     (we use RecursiveCharacterTextSplitter, not HTMLHeaderTextSplitter)
langchain==1.3.9
langchain-community==0.3.31
langchain-core==1.3.3
langchain-text-splitters==1.1.2

# --- Vector Database ---
# 0.5.20 -> 1.5.9: Chroma Cloud runs a newer server version that's no longer
# compatible with the response format this old client expected when listing
# collections (KeyError: '_type' in list_collections(), reproduced and
# confirmed against the real instance). The local ChromaDB Docker image
# (docker-compose.yml) was bumped to the same version for the same reason —
# and its healthcheck endpoint changed from /api/v1/heartbeat (410 Gone on
# 1.5.9) to /api/v2/heartbeat, updated in docker-compose.yml and
# scripts/verify_setup.py.
chromadb==1.5.9              # ChromaDB client (local Docker + Chroma Cloud, same client)
# PYSEC-2026-311: pre-auth RCE in the ChromaDB SERVER (not in this pip
# client) via trust_remote_code=true when creating a collection with a
# remote-model-based embedding function — no patched version yet. Our code
# never passes trust_remote_code (see chromadb_adapter.py), but the real
# mitigation is network-level, not code-level: the local ChromaDB server has
# no auth of its own, so it must not be exposed — see the 127.0.0.1 bind in
# docker-compose.yml. In production we use Chroma Cloud (managed by Chroma,
# outside our direct control), not this image.
